cmake_minimum_required(VERSION 3.24)
project(ThesisEngine VERSION 0.7.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
FetchContent_Declare(glm GIT_REPOSITORY https://github.com/g-truc/glm.git GIT_TAG 1.0.1 GIT_SHALLOW TRUE)
FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG v1.91.8 GIT_SHALLOW TRUE)
FetchContent_Declare(miniaudio GIT_REPOSITORY https://github.com/mackron/miniaudio.git GIT_TAG 0.11.21 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(glfw glm imgui miniaudio)
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
add_library(glad STATIC third_party/glad/src/glad.c)
target_include_directories(glad PUBLIC third_party/glad/include)
add_library(engine_ui STATIC ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
target_include_directories(engine_ui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(engine_ui PUBLIC glfw OpenGL::GL)
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS src/Renderer/*.cpp src/Audio/*.cpp src/Debug/*.cpp src/Physics/*.cpp)
add_library(thesis_engine STATIC ${ENGINE_SOURCES} src/stb_image_impl.cpp)
target_include_directories(thesis_engine PUBLIC src third_party/stb ${miniaudio_SOURCE_DIR})
target_compile_definitions(thesis_engine PUBLIC GLFW_INCLUDE_NONE)
target_link_libraries(thesis_engine PUBLIC glad glm::glm engine_ui Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
 target_link_libraries(thesis_engine PUBLIC winmm ole32)
endif()
add_executable(ThesisDemo src/main.cpp)
target_link_libraries(ThesisDemo PRIVATE thesis_engine)
if(MINGW)
 target_link_options(ThesisDemo PRIVATE -static-libgcc -static-libstdc++)
endif()
add_custom_command(TARGET ThesisDemo POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:ThesisDemo>/assets")
set_property(TARGET ThesisDemo PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:ThesisDemo>")
include(CTest)
if(BUILD_TESTING)
 add_executable(engine_tests tests/engine_tests.cpp)
 target_link_libraries(engine_tests PRIVATE thesis_engine)
 add_test(NAME core COMMAND engine_tests)
 add_test(NAME gui_smoke COMMAND ThesisDemo --smoke-test)
 set_tests_properties(gui_smoke PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:ThesisDemo>" TIMEOUT 30)
endif()
